useState is good with simple value and simple logic 

but useReducer is good with complex logic 


CartContext is very important component 
here we will  update array 

import [createContext, useReducer] from 'react' 

const CartContext = createContext({
    items : [], 
    addItem : (item) => {}
    removeItem : (item) => {}
    
})


function cartReducer(state,action)
{

    action.type 

    state.items  

}

export function CartContextProvider({children})
{

    useReducer(cartReducer)
}
